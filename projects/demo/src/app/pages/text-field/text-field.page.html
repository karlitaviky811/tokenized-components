Uso

// Componente
readonly model = signal({ email: '' });
readonly f = form(this.model);

// Template
<lib-text-field
  label="Email"
  [formField]="f.email"
  appearance="outline"
  placeholder="usuario@dominio.com"
  [fullWidth]="true"
/>
