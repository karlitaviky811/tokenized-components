Estados de feedback

❖ Error con ícono SVG </>

Ingresa menos de 8 dígitos y sal del campo para ver el ícono de error.

error + suffixIcon="info-error" — el ícono SVG aparece sólo cuando hay error activo
❖ Cargando resultado </>

Escribe algo — el spinner aparece mientras se simula la búsqueda (1.5 s).

loading=true — Spinner en sufijo mientras dura la operación async · desaparece al completar

Uso

// TS — validación de longitud exacta en el componente consumidor
readonly idDocValue = signal('');
readonly idDocError = signal<string | null>(null);

onIdValueChange(value: string): void {
  this.idDocValue.set(value);
  this.idDocError.set(null);
}

onIdBlur(): void {
  if (this.idDocValue().length < requiredLength) {
    this.idDocError.set(`Longitud requerida: ${requiredLength} caracteres`);
  }
}

// Template
<lib-text-field
  label="Cédula"
  format="id"
  mask="#-####-####"
  [error]="idDocError()"
  [value]="idDocValue()"
  (valueChange)="onIdValueChange($event)"
  (touch)="onIdBlur()"
/>
