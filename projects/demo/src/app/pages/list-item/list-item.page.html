Uso

interface LibListItemData {
  id: string;
  label: string;
  description?: string;
  avatarText?: string | null;
  avatarImage?: string | null;
  avatarAlt?: string;
  disabled?: boolean;
}

<lib-list
  [items]="items()"
  [selectedIds]="selectedIds"
  (selectionChange)="onSelect($event)"
